## 5. Automatisierte Test-Suite

```rust
// tests/reference_tests.rs

#[cfg(test)]
mod reference_tests {
    use synthesis_engine::*;
    
    struct TestCase {
        name: &'static str,
        year: i32,
        month: u32,
        day: u32,
        hour: u32,
        minute: u32,
        lat: f64,
        lon: f64,
        tz: f64,
        expected_type: &'static str,
        expected_profile: &'static str,
        expected_sun_gate: i32,
    }
    
    const TEST_CASES: &[TestCase] = &[
        TestCase {
            name: "Ra Uru Hu",
            year: 1948, month: 4, day: 28,
            hour: 8, minute: 14,
            lat: 45.5017, lon: -73.5673, tz: -5.0,
            expected_type: "MANIFESTOR",
            expected_profile: "5/1",
            expected_sun_gate: 22,
        },
        // Weitere Testfälle hier...
    ];
    
    #[test]
    fn test_all_reference_cases() {
        let ephe = EphemerisCalculator::new().unwrap();
        
        for case in TEST_CASES {
            let birth_data = BirthData {
                year: case.year,
                month: case.month,
                day: case.day,
                hour: case.hour,
                minute: case.minute,
                latitude: case.lat,
                longitude: case.lon,
                timezone: case.tz,
            };
            
            let chart = calculate_hd_chart_with_ephemeris(birth_data, &ephe)
                .expect(&format!("Berechnung fehlgeschlagen für {}", case.name));
            
            assert_eq!(
                chart.energy_type, case.expected_type,
                "Falscher Energie-Typ für {}", case.name
            );
            
            assert_eq!(
                chart.profile, case.expected_profile,
                "Falsches Profil für {}", case.name
            );
            
            let sun_gate = chart.gates.iter()
                .find(|g| g.planet == "SUN")
                .map(|g| g.number)
                .unwrap_or(0);
                
            assert_eq!(
                sun_gate, case.expected_sun_gate,
                "Falsches Sonnen-Gate für {}", case.name
            );
        }
    }
}
```

```rust
pub fn debug_print_chart(chart: &HumanDesignChart) {
    println!("=== Human Design Chart ===");
    println!("Energie-Typ: {}", chart.energy_type);
    println!("Profil: {}", chart.profile);
    println!("Autorität: {}", chart.authority);
    println!("\nAktivierte Tore:");
    
    for gate in &chart.gates {
        println!(
            "  {}: Gate {}.{}/{}/{}/{} ({})",
            gate.planet,
            gate.number,
            gate.line,
            gate.color,
            gate.tone,
            gate.base,
            if gate.is_design { "Design" } else { "Personality" }
        );
    }
    
    println!("\nDefinierte Zentren: {:?}", chart.defined_centers);
    println!("Kanäle: {:?}", chart.channels.iter().map(|c| format!("{}-{}", c.gate_1, c.gate_2)).collect::<Vec<_>>());
}
```
